106 def ansi_to_html(text):
107 """Convert ANSI colored text to HTML with CSS classes."""
108 result = []
109 pos = 0
110 active_styles = []
111
112 # Process all ANSI escape sequences
113 for match in ANSI_ESCAPE_RE.finditer(text):
114 # Add text before the escape sequence
115 if pos < match.start():
116 if active_styles:
117 css_class = " ".join(active_styles)
118 result.append(f'<span class="{css_class}">{escape(text[pos:match.start()])}</span>')
119 else:
120 result.append(escape(text[pos:match.start()]))
121
122 # Process the escape sequence
123 ansi_codes = parse_ansi_code(match.group(1))
124 if 0 in ansi_codes: # Reset
125 active_styles = []
126 else:
127 active_styles = ansi_to_css(ansi_codes)
128
129 pos = match.end()
130
