47 def dump_command(args):
48 """
49 Process a file with termdump and convert it to HTML.
50 This command:
51 1. Reads input file
52 2. Processes terminal escape sequences with pytermdump
53 3. Converts the processed output to HTML with ansi2html
54 4. Writes the HTML to the output file
55 """
56 # Determine input source
57 if args.input_file == '-':
58 input_data = sys.stdin.buffer.read()
59 else:
60 try:
61 with open(args.input_file, 'rb') as f:
62 input_data = f.read()
63 except Exception as e:
64 print(f"Error reading input file: {e}", file=sys.stderr)
65 sys.exit(1)
66
67 # Process with pytermdump
68 try:
69 processed_data = pytermdump.termdump(input_data)
70 processed_text = processed_data.decode('utf-8', errors='replace')
71 except Exception as e:
72 print(f"Error processing with pytermdump: {e}", file=sys.stderr)
73 sys.exit(1)
74
75 # Convert to HTML
76 try:
77 html_output = generate_html(processed_text)
78 except Exception as e:
79 print(f"Error converting to HTML: {e}", file=sys.stderr)
80 sys.exit(1)
81
82 # Determine output destination
83 if args.output_file == '-':
84 sys.stdout.write(html_output)
85 else:
86 try:
87 with open(args.output_file, 'w') as f:
88 f.write(html_output)
89 print(f"HTML output written to {args.output_file}", file=sys.stderr)
90 except Exception as e:
91 print(f"Error writing output file: {e}", file=sys.stderr)
92 sys.exit(1)

1333

1336 3. Run the `claude` command with all provided arguments

1339 ```bash
